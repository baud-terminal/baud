#!/usr/bin/env bash
# Draws step2/mode.png and b192/table-1-table.png, table-2-psql.png and
# table-3-pipe-in-text.png, each from the scene of the same name beside
# it, through baud's own CPU renderer: panels.rs.txt, built here as a
# crate of its own on the working tree's baud-core, baud-domain,
# baud-font and baud-render, draws each panel and the title, and the
# pinned ImageMagick stacks them in a frame of the scene's colour. The
# face is baud's embedded Regular, baud-app/fonts/built/Regular.ttf.
#
#   bash docs/research/notes/ab/panels.sh.txt [<scene.txt> ...]
#
# With no scene it draws all four. Each panel's order is worked in
# panels.rs.txt, so a panel is the screen its option drew on 2026-10-24,
# whatever today's renderer draws for that text. Drawn again so on
# 2026-11-03 with a placeholder user and a `$` prompt in the sample text:
# the same text with the earlier sample's owner and prompt drew the
# committed pictures of 2026-10-24 again, every cell in the same place,
# the only pixels apart in the cells of three characters whose glyphs
# baud's font has redrawn since: ד, ז (narrowed 2026-10-28) and ₪
# (baud's own from 2026-10-25).
set -euo pipefail

here=$(cd "$(dirname "$0")" && pwd)
root=$(cd "$here/../../../.." && pwd)
NIXPKGS=github:NixOS/nixpkgs/319e4cebef10cbb636c6ea392137b52ad105336a
magick=$(nix build --no-link --print-out-paths "$NIXPKGS#imagemagick")/bin/magick

tmp=$(mktemp -d)
trap 'rm -rf "$tmp"' EXIT
mkdir -p "$tmp/panels/src" "$tmp/out"
cp "$here/panels.rs.txt" "$tmp/panels/src/main.rs"
cat >"$tmp/panels/Cargo.toml" <<EOF
[package]
name = "panels"
version = "0.0.0"
edition = "2024"
publish = false

[dependencies]
baud-core = { path = "$root/baud-core", features = ["alloc"] }
baud-domain = { path = "$root/baud-domain" }
baud-font = { path = "$root/baud-font" }
baud-render = { path = "$root/baud-render" }

[workspace]
EOF
target=${CARGO_TARGET_DIR:-$root/target/panels}
CARGO_TARGET_DIR=$target cargo build --release --quiet --manifest-path "$tmp/panels/Cargo.toml"

if [ $# -eq 0 ]; then
  set -- "$here/step2/mode.txt" "$here/b192/table-1-table.txt" \
    "$here/b192/table-2-psql.txt" "$here/b192/table-3-pipe-in-text.txt"
fi
for scene in "$@"; do
  ppms=()
  while IFS= read -r p; do ppms+=("$p"); done < <(
    "$target/release/panels" "$root/baud-app/fonts/built/Regular.ttf" "$scene" "$tmp/out"
  )
  border=$(sed -n 's/^border //p' "$scene")
  # -strip: no date in the PNG, so a second run writes the same bytes.
  "$magick" "${ppms[@]}" -bordercolor "$border" -border 6 -append -border 6 -strip "${scene%.txt}.png"
  echo "panels: ${scene%.txt}.png, ${#ppms[@]} strips"
done
