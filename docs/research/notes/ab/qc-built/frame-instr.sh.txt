#!/bin/bash
Q=$1; cd $Q
CN="/System/Library/Fonts/Supplemental/Courier New.ttf"; JR=baud-app/fonts/built/Regular.ttf
G=../../docs/research/notes/ab/qc/t-gen5.txt
order=(f1 f2 f2b)
for r in 1 2 3 4 5 6; do
  echo "# round $r load $(sysctl -n vm.loadavg)"
  k=$(( (r-1) % 3 )); arms=("${order[@]:k}" "${order[@]:0:k}")
  for a in "${arms[@]}"; do
    for w in "vim:" "btop:" "cat-cjk:" "gen5-courier:$CN" "gen5-built:$JR"; do
      name=${w%%:*}; faces=${w#*:}; rec=$name; case $name in gen5-*) rec=$G;; esac
      run() { if [ -n "$faces" ]; then FRAME_FACES="$faces" /usr/bin/time -l $Q/bin/frame-$a $rec $1 2>&1; else /usr/bin/time -l $Q/bin/frame-$a $rec $1 2>&1; fi; }
      o1=$(run 10); o2=$(run 50)
      i1=$(echo "$o1" | awk '/instructions retired/{print $1}'); i2=$(echo "$o2" | awk '/instructions retired/{print $1}')
      row=$(echo "$o2" | grep -A1 '^recording' | tail -1 | awk '{$1=""; $3=""; $4=""; $5=""; print}')
      echo "$r $a $name $(python3 -c "print(f'{($i2-$i1)/40/1e6:.3f}')") |$row"
    done
  done
done
