#!/bin/bash
Q=$1; CN="/System/Library/Fonts/Supplemental/Courier New.ttf"; JR=baud-app/fonts/built/Regular.ttf
order=(c1 c2 c1b c2b)
for r in 1 2 3 4 5 6; do
  echo "# round $r load $(sysctl -n vm.loadavg) power $(pmset -g batt | grep -o "'.*'" | head -1)"
  k=$(( (r-1) % 4 )); arms=("${order[@]:k}" "${order[@]:0:k}")
  for a in "${arms[@]}"; do
    for mode in "norm:$CN:10000" "norm:$JR:10000" "bench:$CN:500" "bench:$JR:200"; do
      IFS=: read m f n <<< "$mode"
      v=$($Q/bin/qc-$a $m "$f" 0 $Q/bench-clusters.txt $n)
      echo "$r $a $m $(basename "$f" .ttf | tr ' ' _) $v"
    done
  done
done
